49 bool init() {
50 _bound = false;
51 if (_msc == nullptr) {
52 return false;
53 }
54 uint8_t addr = _want;
55 if (addr == kAnyDevice) {
56 if (_msc->mountedDeviceCount() == 0) {
57 return false;
58 }
59 addr = _msc->deviceAddress(0);
60 }
61 if (!_msc->isMounted(addr) || _msc->blockSize(addr, _lun) != kSectorSize ||
62 _msc->blockCount(addr, _lun) == 0) {
63 return false;
64 }
65 _addr = addr;
66 _generation = _msc->mountGeneration(addr);
67 _bound = true;
68 return true;
69 }

87 bool read(uint32_t lba, void *buffer, uint32_t count) {
88 if (!inRange(lba, count)) {

93 uint16_t n = count > kMaxBlocksPerCall ? kMaxBlocksPerCall : (uint16_t)count;

95 return false;

100 }
