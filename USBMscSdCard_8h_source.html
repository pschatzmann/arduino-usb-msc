1/*
2 * USBMscSdCard -- exports an SD card (SPI or SDIO) over USB device mode.
3 *
4 * Wraps the raw sector access of SdFat's SdCard so the whole card -- with
5 * whatever filesystem is on it (FAT, exFAT, ext2/3/4) -- can be handed to
6 * USBDeviceMSC. The PC sees the card as a USB stick; the board never
7 * interprets the filesystem.
8 *
9 * Needs the "SdFat" library (bundled with the RP2040 core; install
10 * "SdFat - Adafruit Fork" or "SdFat" from the Library Manager on ESP32 etc.).
11 * Use sd.cardBegin() -- not sd.begin() -- so cards without a FAT filesystem
12 * (e.g. ext4) work too.
13 */
14#pragma once
15
16#include <SdFat.h>
17
18#include "USBDeviceMSC.h"
19
20class USBMscSdCard : public USBMscBlockDevice {
21public:
23 explicit USBMscSdCard(SdCard *card) : _card(card) {}
24
25 uint32_t blockCount() override { return _card ? _card->sectorCount() : 0; }
26 bool readBlocks(uint32_t lba, void *buffer, uint32_t count) override {
27 return _card && _card->readSectors(lba, (uint8_t *)buffer, count);
28 }
29 bool writeBlocks(uint32_t lba, const void *buffer, uint32_t count) override {
30 return _card && _card->writeSectors(lba, (const uint8_t *)buffer, count);
31 }
32 void flush() override {
33 if (_card) _card->syncDevice();
34 }
35
36private:
37 SdCard *_card;
38};
