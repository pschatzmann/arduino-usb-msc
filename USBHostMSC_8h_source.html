1/*
2 * usb-msc
3 *
4 * Thin Arduino wrapper around TinyUSB's host-mode Mass Storage Class
5 * driver (tuh_msc_*), giving blocking, SCSI-block-level access to a USB
6 * flash drive attached to the board's USB host port.
7 *
8 * Built on TinyUSB: the copy that is built into the arduino-esp32 core
9 * (ESP32-S2/S3), the one bundled with the RP2040 core, or otherwise the
10 * "Adafruit TinyUSB Library". USB Host mode must be selected for the
11 * target board:
12 * - RP2040 (earlephilhower core): Tools > USB Stack >
13 * "Adafruit TinyUSB Host (native)"
14 * - ESP32-S2/S3 (arduino-esp32 core >= 3.3): uses the TinyUSB that is
15 * built into the core -- nothing else to install. Tools > USB Mode >
16 * "USB-OTG (TinyUSB)". Serial is then UART0 (see docs/SINGLE_PORT.md).
17 * - STM32 (STM32duino core, F4/F7 OTG_FS parts only -- see
18 * USBHostMSC_stm32.cpp): build with -DCFG_TUH_ENABLED=1, and apply the
19 * small Adafruit_TinyUSB_Arduino patch described in
20 * docs/STM32.md (host mode is off by default there and needs two local
21 * changes to that library to enable safely)
22 * - ESP32-P4 (write-back cache) and anything built with
23 * -DUSBHOSTMSC_USE_ADAFRUIT_TINYUSB use the "Adafruit TinyUSB Library"
24 * instead, which needs a small patch for its native host controller --
25 * see docs/ESP32_NATIVE.md.
26 * - MCUs with no USB peripheral at all (classic ESP32, ESP32-C3/C6/H2,
27 * SAMD, nRF52, ...) can still act as a USB host via an external
28 * MAX3421E host-shield chip wired over SPI -- see docs/MAX3421E.md.
29 */

32// Which TinyUSB stack do we sit on?
33// - ESP32-S2/S3 (arduino-esp32 >= 3.3): the TinyUSB host stack that is
34// already compiled into the core, started with USBHost.begin().
35// - everything else (RP2040 core, STM32, MAX3421E boards, ESP32-P4, or
36// -DUSBHOSTMSC_USE_ADAFRUIT_TINYUSB): the "Adafruit TinyUSB Library".

103// Root hub port used to bring up the host controller. Native (non-MAX3421)
104// ports only expose one usable value; override with a build flag if needed.
105// MAX3421E boards use rhport 1 by convention (see Adafruit's own examples
106// and docs/MAX3421E.md) -- rhport 0 there is reserved for a chip's own
107// native USB peripheral, if it has one and it's used for device mode.
108#ifndef USB_HOST_MSC_RHPORT
109#if defined(BOARD_TUH_RHPORT)
110#define USB_HOST_MSC_RHPORT BOARD_TUH_RHPORT
111#elif defined(CFG_TUH_MAX3421) && CFG_TUH_MAX3421
112#define USB_HOST_MSC_RHPORT 1
113#else
114#define USB_HOST_MSC_RHPORT 0
115#endif
116#endif

118// Upper bound for one blocking SCSI command (inquiry/read/write). If the
119// drive is unplugged or stops answering mid-command, the call returns
120// false after this long instead of spinning forever. Override with a build
121// flag for very slow media.
122#ifndef USB_HOST_MSC_IO_TIMEOUT_MS
123#define USB_HOST_MSC_IO_TIMEOUT_MS 10000
124#endif
125
126#if USBHOSTMSC_ESP32_CORE
127// ESP32 core path: SCSI transfers go through a DMA-capable bounce buffer in
128// chunks of at most this many blocks. 1 is the same conservative value the
129// core's own USBHostMSC uses (some controllers/sticks wedge on longer
130// transfers); raise it with a build flag if your drive copes, for speed.
131#ifndef USB_HOST_MSC_MAX_BLOCKS_PER_XFER
132#define USB_HOST_MSC_MAX_BLOCKS_PER_XFER 1
133#endif
134// Pause after each written chunk (ms); some sticks wedge when hammered.
135#ifndef USB_HOST_MSC_WRITE_PACING_MS
136#define USB_HOST_MSC_WRITE_PACING_MS 1
137#endif
138#endif
