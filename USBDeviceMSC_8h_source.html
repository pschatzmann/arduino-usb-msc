1/*
2 * usb-msc -- USB *device* mode Mass Storage
3 *
4 * The counterpart of USBHostMSC: instead of reading a USB stick, the board
5 * *is* one. A PC that is plugged into the board's USB port sees a drive
6 * whose sectors are served by a block driver of TinyFATFS or TinyEXT4
7 * (RamIO, ArduinoSpiIO, Esp32SdmmcIO, NBDClientIO, ...) or by your own class
8 * with the same disk_read()/disk_write()/disk_ioctl() interface.
9 *
10 * Built on the device-mode MSC class that is already part of the platform:
11 * - ESP32-S2/S3 (arduino-esp32 core): the core's USBMSC. Select
12 * Tools > USB Mode > "USB-OTG (TinyUSB)".
13 * - everything else (RP2040, SAMD, nRF52, STM32, ...): Adafruit_USBD_MSC
14 * from the "Adafruit TinyUSB Library". On RP2040 select
15 * Tools > USB Stack > "Adafruit TinyUSB" (the *device* stack, not
16 * "...Host (native)").
17 *
18 * Device and host mode need separate USB controllers; this class and
19 * USBHostMSC can only run at the same time on boards that have two
20 * (e.g. RP2040 with PIO-USB for the host side, or a MAX3421E host chip).
21 *
22 * Only one USBDeviceMSC instance (a single LUN) is supported.
23 *
24 * Do not let the MCU modify a filesystem on the exported medium while the
25 * PC has it mounted -- the PC caches aggressively and both sides would
26 * corrupt each other. Use USBDeviceMSC::mediaPresent(false) to "eject"
27 * while the MCU accesses the data, then mediaPresent(true) again.
28 */

95 template <class IO> bool begin(IO &io, bool startUsb = true) {
96 using Sig = decltype(sigOf(&IO::disk_read));
97 using Cmd = decltype(cmdOf(&IO::disk_ioctl));
98 if (io.disk_initialize(0) & (kNoInit | kNoDisk)) {
99 return false;
100 }
101 typename Sig::Lba sectors = 0;
102 if ((int)io.disk_ioctl(0, (Cmd)kGetSectorCount, &sectors) != 0 ||
103 sectors == 0) {
104 return false;
105 }
106 uint16_t size = 0; // GET_SECTOR_SIZE is optional (512 if unsupported)
107 if ((int)io.disk_ioctl(0, (Cmd)kGetSectorSize, &size) == 0 && size != 0 &&
108 size != kSectorSize) {
109 return false;
110 }
111 Disk d;
112 d.ctx = &io;
113 d.read = [](void *ctx, uint32_t lba, void *buf, uint32_t n) {
114 return (int)((IO *)ctx)->disk_read(0, (typename Sig::Buf)buf,
115 (typename Sig::Lba)lba,
116 (typename Sig::Count)n) == 0;
117 };
118 d.write = [](void *ctx, uint32_t lba, const void *buf, uint32_t n) {
119 return (int)((IO *)ctx)->disk_write(0, (const uint8_t *)buf,
120 (typename Sig::Lba)lba,
121 (typename Sig::Count)n) == 0;
122 };
123 d.flush = [](void *ctx) {
124 ((IO *)ctx)->disk_ioctl(0, (Cmd)kCtrlSync, nullptr);
125 };
126 d.sectors = (uint32_t)sectors;
127 return _begin(d, startUsb);
128 }
